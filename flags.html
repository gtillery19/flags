Flag Viewer

viewof selected_flag = Inputs.select(flag_files, {
  label: "Select a flag:",
  value: "us.png"
})

viewof match_type = Inputs.radio(
  ["Color", "Layout", "History of Design"],
  { label: "Match by:", value: "Color" }
)

selected_country = country_names[selected_flag.replace(".png", "")]

html`
  <div>
    <h3>Selected Flag</h3>
    <img src="w2560/${selected_flag}" style="max-width: 400px; border: 1px solid #ccc;">
    <p style="font-size: 1.25rem; font-weight: bold; margin-top: 0.5rem;">${selected_country}</p>
  </div>
`
lookup = match_type === "Color"
  ? similar_color
  : match_type === "Layout"
  ? similar_layout
  : similar_history

similar_flag = lookup[selected_flag]
similar_country = country_names[similar_flag.replace(".png", "")]
history_text = match_type === "History of Design" ? history_notes[selected_flag] : ""

html`
  <div>
    <h3>Most Similar Flag (${match_type})</h3>
    <img src="w2560/${similar_flag}" style="max-width: 400px; border: 1px solid #ccc;">
    <p style="font-size: 1.25rem; font-weight: bold; margin-top: 0.5rem;">${similar_country} (${similar_flag})</p>
    <p style="max-width: 500px; color: #444;">${history_text}</p>
  </div>
`
